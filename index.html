JSscript.js
const cards = document.querySelectorAll(
  ".beam-card"
);
cards.forEach((card) => {
  card.addEventListener("pointermove", e => {
    const rect = card.getBoundingClientRect();
    card.style.setProperty("--x",
      e.clientX - rect.left + "px");
  });
});
